// 1. Constantes y Variables
pi = 3.14159265;
angulo = pi / 4;

// 2. Funciones Trigonométricas
s = sin(angulo);
c = cos(angulo);
t = tan(angulo);

// 3. Operaciones Aritméticas Combinadas (+, -, *, /, %)
potencia_seno = s * s;
potencia_coseno = c * c;
suma_cuadrados = potencia_seno + potencia_coseno;

// 4. Valor Absoluto, Módulo y Raíz Cuadrada
error = abs(t - 1.0);
residuo = 100 % 30;
modulo_vectorial = sqrt(suma_cuadrados);

// 5. Expresión Final
resultado = (suma_cuadrados * 10) + residuo - error + modulo_vectorial;
resultado;
