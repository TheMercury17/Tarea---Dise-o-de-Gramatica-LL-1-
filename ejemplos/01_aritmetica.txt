a = 15;
b = 4;
suma = a + b;
resta = a - b;
producto = a * b;
division = a / b;

// Evaluación con precedencia de operadores (* antes de +)
precedencia = a + b * 2;
agrupado = (a + b) * 2;

// Menos unario y expresiones compuestas
negativo = - (a * 2);
final = precedencia + agrupado + negativo;
final;
